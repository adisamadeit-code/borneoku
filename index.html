-- 1. Hapus tabel lama jika ada DROP TABLE IF EXISTS payouts CASCADE; DROP TABLE IF EXISTS order_items CASCADE; DROP TABLE IF EXISTS products CASCADE; DROP TABLE IF EXISTS stores CASCADE; -- 2. Buat tabel stores CREATE TABLE stores ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, owner_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, name VARCHAR(255), slug VARCHAR(255), category VARCHAR(100), description TEXT, logo_url TEXT, banner_url TEXT, phone_number VARCHAR(20), address TEXT, status VARCHAR(20) DEFAULT 'approved', commission_rate NUMERIC(5,2) DEFAULT 5.00, bank_name VARCHAR(100), account_number VARCHAR(50), account_holder_name VARCHAR(255), created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); -- 3. Buat tabel products (dengan penambahan kolom status & title agar kompatibel) CREATE TABLE products ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, store_id UUID REFERENCES stores(id) ON DELETE SET NULL, name VARCHAR(255), title VARCHAR(255), slug VARCHAR(255), description TEXT, price NUMERIC(12,2), stock INT DEFAULT 0, unit VARCHAR(50) DEFAULT 'pcs', product_type VARCHAR(50) DEFAULT 'physical', image_url TEXT, status VARCHAR(20) DEFAULT 'approved', is_available BOOLEAN DEFAULT true, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); -- 4. Buat tabel order_items CREATE TABLE order_items ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, order_id UUID, store_id UUID REFERENCES stores(id) ON DELETE SET NULL, product_id UUID REFERENCES products(id) ON DELETE SET NULL, quantity INT NOT NULL DEFAULT 1, unit_price NUMERIC(12,2), gross_amount NUMERIC(12,2), commission_rate_applied NUMERIC(5,2), platform_fee NUMERIC(12,2), merchant_net_amount NUMERIC(12,2), created_at TIMESTAMPTZ DEFAULT NOW() ); -- 5. Buat tabel payouts CREATE TABLE payouts ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, store_id UUID REFERENCES stores(id) ON DELETE CASCADE, amount NUMERIC(12,2), status VARCHAR(20) DEFAULT 'pending', payout_date TIMESTAMPTZ, proof_url TEXT, notes TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 6. Matikan RLS agar frontend bisa membaca & menulis tanpa kendala izin ALTER TABLE stores DISABLE ROW LEVEL SECURITY; ALTER TABLE products DISABLE ROW LEVEL SECURITY; ALTER TABLE order_items DISABLE ROW LEVEL SECURITY; ALTER TABLE payouts DISABLE ROW LEVEL SECURITY; -- 7.
